create role anon nologin;
create role authenticated nologin;
create role supabase_auth_admin nologin;
create schema auth;
grant usage on schema auth to anon, authenticated, supabase_auth_admin;
grant usage on schema public to anon, authenticated;
create table auth.users (id uuid primary key default gen_random_uuid(), email text unique, raw_user_meta_data jsonb default '{}'::jsonb);
create function auth.uid() returns uuid language sql stable as $$
  select nullif(current_setting('request.jwt.claims', true)::json ->> 'sub', '')::uuid $$;
grant execute on function auth.uid() to anon, authenticated;

\set ON_ERROR_STOP 1
\pset footer off
\set QUIET on
create schema prueba;
create or replace function prueba.como(correo text) returns void language sql as $$
  select set_config('request.jwt.claims', json_build_object('sub', (select id from auth.users where email = correo), 'role','authenticated')::text, true) $$;

\echo '=== A. Movimientos de prueba ANTES de la carga (referencian productos de prueba)'
\echo '--- A1. Trabajador: cotización enviada a cliente@cse.test con DVR DS-7204HGHI-M1(C) (está en el Excel) y EZVIZ CS-H6C (no está)'
begin; do $$begin perform prueba.como('trabajador@cse.test'); end$$; set local role authenticated;
select numero, estado, total from public.guardar_borrador_cotizacion(null,
  (select id from public.clientes where correo = 'cliente@cse.test'),
  'instalacion', 'Cámaras para la casa', 0, true, null,
  (select jsonb_agg(jsonb_build_object('producto_id', id, 'cantidad', 1))
   from public.productos where codigo in ('DS-7204HGHI-M1(C)', 'CS-H6C')));
update public.cotizaciones set estado = 'nueva' where estado = 'borrador' and descripcion_necesidad = 'Cámaras para la casa';
commit;

\echo '--- A2. Cliente: pedido desde el carrito con fuente PRUEBA-FUE-12V2A (no está en el Excel) y cámara DS-2CE16D0T-IRPF (sí está)'
begin; do $$begin perform prueba.como('cliente@cse.test'); end$$; set local role authenticated;
select numero, origen, total from public.registrar_pedido_carrito(
  (select jsonb_agg(jsonb_build_object('producto_id', id, 'cantidad', 2))
   from public.productos_catalogo where codigo in ('PRUEBA-FUE-12V2A', 'DS-2CE16D0T-IRPF')),
  false, null);
commit;

\echo '--- A3. Cliente: solicitud de cotización con producto de interés EZVIZ CS-H9C (no está en el Excel)'
begin; do $$begin perform prueba.como('cliente@cse.test'); end$$; set local role authenticated;
select numero, estado from public.crear_solicitud_cotizacion('mantenimiento', 'Revisar cámara exterior',
  (select jsonb_build_array(jsonb_build_object('producto_id', id, 'cantidad', 1))
   from public.productos_catalogo where codigo = 'CS-H9C'));
commit;

\echo '--- A4. Cliente acepta la cotización A1: pedido desde cotización'
begin; do $$begin perform prueba.como('cliente@cse.test'); end$$; set local role authenticated;
select numero, origen, total from public.aceptar_cotizacion(
  (select id from public.cotizaciones where descripcion_necesidad = 'Cámaras para la casa'));
commit;

\echo '=== B. Foto de cotizaciones, pedidos y líneas antes de la carga'
create table prueba.antes_cot as
  select id, numero, cliente_id, estado, subtotal, descuento, impuesto, costo_servicio, total from public.cotizaciones;
create table prueba.antes_cot_lin as
  select id, cotizacion_id, producto_id, codigo_producto, nombre_producto, precio_unitario, cantidad from public.cotizacion_lineas;
create table prueba.antes_interes as
  select id, cotizacion_id, producto_id, nombre_producto, cantidad from public.cotizacion_productos_interes;
create table prueba.antes_ped as
  select id, numero, cliente_id, origen, cotizacion_id, subtotal, impuesto, total from public.pedidos;
create table prueba.antes_ped_lin as
  select id, pedido_id, producto_id, codigo_producto, nombre_producto, precio_unitario, cantidad from public.pedido_lineas;
create table prueba.antes_prod as select id, codigo, nombre, es_prueba from public.productos;

select (select count(*) from prueba.antes_cot) as cotizaciones,
       (select count(*) from prueba.antes_cot_lin) as lineas_cot,
       (select count(*) from prueba.antes_interes) as productos_interes,
       (select count(*) from prueba.antes_ped) as pedidos,
       (select count(*) from prueba.antes_ped_lin) as lineas_ped,
       (select count(*) from public.productos where es_prueba) as productos_prueba;

\echo 'Productos de prueba usados en cotizaciones, pedidos o productos de interés:'
select distinct p.codigo, p.nombre from public.productos p
where p.id in (select producto_id from public.cotizacion_lineas
               union select producto_id from public.pedido_lineas
               union select producto_id from public.cotizacion_productos_interes)
order by 1;

\echo '=== D. Comprobaciones después de la carga'
\echo '--- D1. Códigos repetidos (esperado: 0 filas)'
select upper(trim(codigo)) as codigo, count(*) from public.productos
where codigo is not null group by 1 having count(*) > 1;

\echo '--- D2. Inventarios negativos (esperado: 0)'
select count(*) as negativos from public.productos where inventario < 0;

\echo '--- D3. Productos por estado (esperado: 76 activos del Excel, 0 de prueba activos)'
select es_prueba, activo, count(*) from public.productos group by 1, 2 order by 1, 2;

\echo '--- D4. Productos de prueba que quedaron (desactivados porque tienen referencias)'
select codigo, nombre, activo from public.productos where es_prueba order by codigo;

\echo '--- D5. Productos de prueba borrados (no estaban en el Excel ni tenían referencias)'
select coalesce(a.codigo, '(sin código)') as codigo, a.nombre from prueba.antes_prod a
where a.es_prueba and not exists (select 1 from public.productos p where p.id = a.id) order by 1;

\echo '--- D6. Productos de prueba que ahora son del Excel (mismo id, es_prueba = false)'
select p.codigo, p.inventario, p.es_prueba, p.activo from prueba.antes_prod a
join public.productos p on p.id = a.id
where a.es_prueba and not p.es_prueba order by 1;

\echo '--- D7. DVR DS-7204HGHI-M1(C) para CP-07 (esperado: inventario 2, activo, mismo id)'
select p.codigo, p.nombre, p.inventario, p.activo, p.es_prueba,
       p.id = (select id from prueba.antes_prod where codigo = 'DS-7204HGHI-M1(C)') as mismo_id
from public.productos p where p.codigo = 'DS-7204HGHI-M1(C)';

\echo '--- D8. Cotizaciones, pedidos y líneas intactos (esperado: 0 diferencias en cada uno)'
select
  (select count(*) from (select * from prueba.antes_cot except
     select id, numero, cliente_id, estado, subtotal, descuento, impuesto, costo_servicio, total from public.cotizaciones) d) as dif_cotizaciones,
  (select count(*) from (select * from prueba.antes_cot_lin except
     select id, cotizacion_id, producto_id, codigo_producto, nombre_producto, precio_unitario, cantidad from public.cotizacion_lineas) d) as dif_lineas_cot,
  (select count(*) from (select * from prueba.antes_interes except
     select id, cotizacion_id, producto_id, nombre_producto, cantidad from public.cotizacion_productos_interes) d) as dif_interes,
  (select count(*) from (select * from prueba.antes_ped except
     select id, numero, cliente_id, origen, cotizacion_id, subtotal, impuesto, total from public.pedidos) d) as dif_pedidos,
  (select count(*) from (select * from prueba.antes_ped_lin except
     select id, pedido_id, producto_id, codigo_producto, nombre_producto, precio_unitario, cantidad from public.pedido_lineas) d) as dif_lineas_ped,
  (select count(*) from public.cotizaciones) as cotizaciones,
  (select count(*) from public.pedidos) as pedidos;

\echo '--- D9. Muestra de la carga: lotes sumados, SIN-COD y precio instalador nulo'
select codigo, marca, categoria, costo, precio_instalador, precio_cliente_final, inventario
from public.productos

order by codigo;
